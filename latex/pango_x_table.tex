\usepackage[left=2cm, right=2cm, top=1cm, bottom=2cm]{geometry}
\usepackage{booktabs}

% Captions of the generated long tables (see pango_x_table/make_table.py):
% footnotesize and full width, laid out like the float captions. The
% continuation head and foot macros take the number of columns in the table.
\makeatletter
\def\LT@makecaption#1#2#3{%
  \LT@mcol\LT@cols c{\hbox to\z@{\hss\parbox[t]\textwidth{%
    \reset@font\footnotesize
    #1{#2: }#3%
    \endgraf\vskip\baselineskip}%
  \hss}}}
\makeatother
\newcommand{\LTcontinuedhead}[1]{%
  \multicolumn{#1}{c}{\footnotesize\tablename~\thetable\ \emph{(continued)}}}
\newcommand{\LTcontinuedfoot}[1]{%
  \multicolumn{#1}{r}{\footnotesize Continued on next page}}
\pagestyle{empty}
